# ===== api/archive.py :: get_tag =====
def get_tag(username: str, tag_id: int) -> dict | None:
    con = _connect(username)
    try:
        r = con.execute("SELECT * FROM archive_tags WHERE id=?", (tag_id,)).fetchone()
        return dict(r) if r else None
    finally:
        con.close()




# ===== api/cup_matches.py :: team_ids_for =====
def team_ids_for(db_name: str, code: str) -> list:
    """우리 DB 팀명 → 스코어맨 팀번호(시즌마다 따로 저장돼도 번호는 같다)."""
    path = schedule_db()
    if not os.path.exists(path):
        return []
    con = sqlite3.connect(path)
    try:
        return [r[0] for r in con.execute(
            "SELECT DISTINCT team_id FROM sm_teams WHERE db_name = ? AND code = ?", (db_name, code))]
    finally:
        con.close()




# ===== api/cup_matches.py :: matches_around =====
def matches_around(team_ids: list, center: datetime, days: int = 10) -> list:
    """그 팀이 center 기준 앞뒤 days일 안에 뛴(뛸) 리그 외 경기 — 시각 순."""
    path = schedule_db()
    if not team_ids or not os.path.exists(path):
        return []
    lo = (center - timedelta(days=days)).strftime("%Y-%m-%d %H:%M")
    hi = (center + timedelta(days=days)).strftime("%Y-%m-%d %H:%M")
    ph = ",".join("?" for _ in team_ids)
    con = sqlite3.connect(path)
    con.row_factory = sqlite3.Row
    try:
        rows = con.execute(
            f"SELECT * FROM cup_matches WHERE (home_id IN ({ph}) OR away_id IN ({ph})) "
            f"AND kickoff BETWEEN ? AND ? ORDER BY kickoff",
            (*team_ids, *team_ids, lo, hi)).fetchall()
        return [dict(r) for r in rows]
    finally:
        con.close()




# ===== api/my_picks.py :: migrate_sample_note_direction =====
def migrate_sample_note_direction(username: str, old: str, new: str) -> int:
    """방향성 선택지 이름이 바뀌었을 때(예: '크로스'→'엇갈림', 2026-09-20) 그 계정에
    이미 저장된 값을 전부 갈아 끼운다. 일회성 마이그레이션 스크립트에서만 부른다 —
    화면 코드 경로에는 없다. 바뀐 행 수를 돌려준다."""
    con = _connect(username)
    try:
        _ensure_sample_notes(con)
        cur = con.execute(
            "UPDATE sample_notes SET direction = ?, updated_dt = datetime('now') WHERE direction = ?",
            (new, old),
        )
        con.commit()
        return cur.rowcount
    finally:
        con.close()
